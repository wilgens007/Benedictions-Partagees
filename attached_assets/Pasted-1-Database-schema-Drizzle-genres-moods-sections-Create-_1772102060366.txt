1) Database schema (Drizzle) — genres + moods + sections

Create/update:

src/db/schema.ts
import { sqliteTable, text, integer } from "drizzle-orm/sqlite-core";
import { sql } from "drizzle-orm";

export const genres = sqliteTable("genres", {
  id: text("id").primaryKey(),
  name: text("name").notNull(),
  slug: text("slug").notNull().unique(),
  order: integer("order").notNull().default(0),
});

export const moods = sqliteTable("moods", {
  id: text("id").primaryKey(),
  name: text("name").notNull(),
  slug: text("slug").notNull().unique(),
  order: integer("order").notNull().default(0),
});

export const albums = sqliteTable("albums", {
  id: text("id").primaryKey(),
  title: text("title").notNull(),
  artist: text("artist"),
  coverUrl: text("cover_url"),
  createdAt: integer("created_at").notNull().default(sql`(unixepoch())`),
});

export const songs = sqliteTable("songs", {
  id: text("id").primaryKey(),
  title: text("title").notNull(),
  artist: text("artist").notNull(),
  albumId: text("album_id").references(() => albums.id),
  audioUrl: text("audio_url").notNull(),
  coverUrl: text("cover_url"),
  language: text("language"), // "ht" | "fr" | "en"
  durationSec: integer("duration_sec"),
  playsCount: integer("plays_count").notNull().default(0),
  createdAt: integer("created_at").notNull().default(sql`(unixepoch())`),
});

// Many-to-many: a song can be in multiple genres
export const songGenres = sqliteTable("song_genres", {
  songId: text("song_id").notNull().references(() => songs.id, { onDelete: "cascade" }),
  genreId: text("genre_id").notNull().references(() => genres.id, { onDelete: "cascade" }),
});

// Many-to-many: a song can be tagged with multiple moods
export const songMoods = sqliteTable("song_moods", {
  songId: text("song_id").notNull().references(() => songs.id, { onDelete: "cascade" }),
  moodId: text("mood_id").notNull().references(() => moods.id, { onDelete: "cascade" }),
});

C) Songs query endpoint (genre/mood/search/sort)

src/app/api/songs/route.ts

import { NextResponse } from "next/server";
import { db } from "@/db/client";
import { songs, songGenres, genres, songMoods, moods } from "@/db/schema";
import { and, desc, eq, like, or, sql } from "drizzle-orm";

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);

  const q = (searchParams.get("q") || "").trim();
  const genreSlug = (searchParams.get("genre") || "").trim();   // e.g. "kompa-gospel"
  const moodSlug = (searchParams.get("mood") || "").trim();     // e.g. "prayer"
  const sort = (searchParams.get("sort") || "new").trim();      // new | popular | az
  const limit = Math.min(parseInt(searchParams.get("limit") || "50", 10), 100);

  // Build WHERE
  const where = [];
  if (q) {
    const pattern = `%${q}%`;
    where.push(or(like(songs.title, pattern), like(songs.artist, pattern)));
  }

  // Base select
  let query = db
    .select({
      id: songs.id,
      title: songs.title,
      artist: songs.artist,
      albumId: songs.albumId,
      audioUrl: songs.audioUrl,
      coverUrl: songs.coverUrl,
      language: songs.language,
      durationSec: songs.durationSec,
      playsCount: songs.playsCount,
      createdAt: songs.createdAt,
    })
    .from(songs);

  // Optional joins for genre/mood filtering
  if (genreSlug) {
    query = query
      .innerJoin(songGenres, eq(songGenres.songId, songs.id))
      .innerJoin(genres, eq(genres.id, songGenres.genreId))
      .where(and(eq(genres.slug, genreSlug), where.length ? and(...where) : sql`1=1`));
  } else if (moodSlug) {
    query = query
      .innerJoin(songMoods, eq(songMoods.songId, songs.id))
      .innerJoin(moods, eq(moods.id, songMoods.moodId))
      .where(and(eq(moods.slug, moodSlug), where.length ? and(...where) : sql`1=1`));
  } else if (where.length) {
    query = query.where(and(...where));
  }

  // Sort
  if (sort === "popular") query = query.orderBy(desc(songs.playsCount), desc(songs.createdAt));
  else if (sort === "az") query = query.orderBy(songs.title);
  else query = query.orderBy(desc(songs.createdAt));

  const rows = await query.limit(limit);
  return NextResponse.json(rows);
}

src/app/music/page.tsx
"use client";

import { useEffect, useMemo, useState } from "react";

type Genre = { id: string; name: string; slug: string; order: number };
type Mood = { id: string; name: string; slug: string; order: number };
type Song = {
  id: string;
  title: string;
  artist: string;
  audioUrl: string;
  coverUrl?: string | null;
  playsCount: number;
  createdAt: number;
};

async function fetchJSON<T>(url: string): Promise<T> {
  const res = await fetch(url, { cache: "no-store" });
  if (!res.ok) throw new Error(`Failed: ${url}`);
  return res.json();
}

function Chip({
  active,
  children,
  onClick,
}: {
  active?: boolean;
  children: React.ReactNode;
  onClick: () => void;
}) {
  return (
    <button
      onClick={onClick}
      className={[
        "px-3 py-1 rounded-full text-sm border",
        active ? "bg-black text-white" : "bg-white text-black",
      ].join(" ")}
    >
      {children}
    </button>
  );
}

function SongCard({ song, onPlay }: { song: Song; onPlay: (s: Song) => void }) {
  return (
    <div className="w-44 shrink-0">
      <button onClick={() => onPlay(song)} className="w-full text-left">
        <div className="aspect-square rounded-xl bg-gray-100 overflow-hidden">
          {song.coverUrl ? (
            // eslint-disable-next-line @next/next/no-img-element
            <img src={song.coverUrl} alt="" className="w-full h-full object-cover" />
          ) : (
            <div className="w-full h-full flex items-center justify-center text-xs text-gray-500">
              No cover
            </div>
          )}
        </div>
        <div className="mt-2">
          <div className="text-sm font-semibold line-clamp-1">{song.title}</div>
          <div className="text-xs text-gray-600 line-clamp-1">{song.artist}</div>
        </div>
      </button>
    </div>
  );
}

function Row({
  title,
  songs,
  onPlay,
}: {
  title: string;
  songs: Song[];
  onPlay: (s: Song) => void;
}) {
  if (!songs.length) return null;
  return (
    <section className="mt-6">
      <div className="flex items-center justify-between">
        <h2 className="text-lg font-bold">{title}</h2>
      </div>
      <div className="mt-3 flex gap-3 overflow-x-auto pb-2">
        {songs.map((s) => (
          <SongCard key={s.id} song={s} onPlay={onPlay} />
        ))}
      </div>
    </section>
  );
}

export default function MusicPage() {
  const [genres, setGenres] = useState<Genre[]>([]);
  const [moods, setMoods] = useState<Mood[]>([]);
  const [activeGenre, setActiveGenre] = useState<string>(""); // slug
  const [activeMood, setActiveMood] = useState<string>(""); // slug
  const [q, setQ] = useState("");
  const [sort, setSort] = useState<"new" | "popular" | "az">("new");

  const [featured, setFeatured] = useState<Song[]>([]);
  const [newReleases, setNewReleases] = useState<Song[]>([]);
  const [popular, setPopular] = useState<Song[]>([]);
  const [filtered, setFiltered] = useState<Song[]>([]);

  // Simple “Now playing” UI hook (swap with your player)
  const onPlay = (song: Song) => {
    // connect to your global audio player store if you have one
    alert(`Play: ${song.title} — ${song.artist}`);
  };

  useEffect(() => {
    (async () => {
      const [g, m] = await Promise.all([
        fetchJSON<Genre[]>("/api/genres"),
        fetchJSON<Mood[]>("/api/moods"),
      ]);
      setGenres(g);
      setMoods(m);
    })().catch(console.error);
  }, []);

  // Load base sections
  useEffect(() => {
    (async () => {
      const [feat, newest, pop] = await Promise.all([
        fetchJSON<Song[]>("/api/songs?sort=new&limit=12"),
        fetchJSON<Song[]>("/api/songs?sort=new&limit=12"),
        fetchJSON<Song[]>("/api/songs?sort=popular&limit=12"),
      ]);
      // For now Featured = newest; later you can add a "featured" flag in DB.
      setFeatured(feat.slice(0, 8));
      setNewReleases(newest);
      setPopular(pop);
    })().catch(console.error);
  }, []);

  // Load filtered list (genre/mood/search/sort)
  useEffect(() => {
    (async () => {
      const params = new URLSearchParams();
      if (q.trim()) params.set("q", q.trim());
      if (activeGenre) params.set("genre", activeGenre);
      if (activeMood) params.set("mood", activeMood);
      params.set("sort", sort);
      params.set("limit", "40");
      const url = `/api/songs?${params.toString()}`;
      const rows = await fetchJSON<Song[]>(url);
      setFiltered(rows);
    })().catch(console.error);
  }, [q, activeGenre, activeMood, sort]);

  const modeLabel = useMemo(() => {
    if (activeGenre) return `Genre: ${genres.find((g) => g.slug === activeGenre)?.name || activeGenre}`;
    if (activeMood) return `Mood: ${moods.find((m) => m.slug === activeMood)?.name || activeMood}`;
    if (q.trim()) return `Search results`;
    return `Browse all`;
  }, [activeGenre, activeMood, q, genres, moods]);

  return (
    <div className="max-w-6xl mx-auto px-4 py-6">
      <div className="flex items-start justify-between gap-4 flex-wrap">
        <div>
          <h1 className="text-2xl font-extrabold">Music</h1>
          <p className="text-sm text-gray-600 mt-1">
            Browse by genre, mood, or search — then press play.
          </p>
        </div>

        <div className="flex gap-2 items-center">
          <select
            className="border rounded-lg px-3 py-2 text-sm"
            value={sort}
            onChange={(e) => setSort(e.target.value as any)}
          >
            <option value="new">Newest</option>
            <option value="popular">Popular</option>
            <option value="az">A–Z</option>
          </select>

          <input
            className="border rounded-lg px-3 py-2 text-sm w-64"
            placeholder="Search song or artist…"
            value={q}
            onChange={(e) => setQ(e.target.value)}
          />
        </div>
      </div>

      {/* Genre chips */}
      <section className="mt-5">
        <div className="text-sm font-semibold mb-2">Genres</div>
        <div className="flex gap-2 flex-wrap">
          <Chip
            active={!activeGenre}
            onClick={() => {
              setActiveGenre("");
              setActiveMood("");
            }}
          >
            All
          </Chip>
          {genres.map((g) => (
            <Chip
              key={g.id}
              active={activeGenre === g.slug}
              onClick={() => {
                setActiveGenre(g.slug);
                setActiveMood("");
              }}
            >
              {g.name}
            </Chip>
          ))}
        </div>
      </section>

      {/* Mood chips */}
      <section className="mt-4">
        <div className="text-sm font-semibold mb-2">Moods</div>
        <div className="flex gap-2 flex-wrap">
          <Chip
            active={!activeMood}
            onClick={() => {
              setActiveMood("");
            }}
          >
            Any
          </Chip>
          {moods.map((m) => (
            <Chip
              key={m.id}
              active={activeMood === m.slug}
              onClick={() => {
                setActiveMood(m.slug);
                setActiveGenre("");
              }}
            >
              {m.name}
            </Chip>
          ))}
        </div>
      </section>

      {/* Sections */}
      {!activeGenre && !activeMood && !q.trim() && (
        <>
          <Row title="Featured Today" songs={featured} onPlay={onPlay} />
          <Row title="New Releases" songs={newReleases} onPlay={onPlay} />
          <Row title="Most Played" songs={popular} onPlay={onPlay} />
        </>
      )}

      {/* Filtered list */}
      <section className="mt-8">
        <div className="flex items-center justify-between">
          <h2 className="text-lg font-bold">{modeLabel}</h2>
          <button
            className="text-sm underline"
            onClick={() => {
              setActiveGenre("");
              setActiveMood("");
              setQ("");
              setSort("new");
            }}
          >
            Reset
          </button>
        </div>

        <div className="mt-4 grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
          {filtered.map((s) => (
            <SongCard key={s.id} song={s} onPlay={onPlay} />
          ))}
        </div>
      </section>
    </div>
  );
}
