🔹 STEP 5 — HOME PAGE (IMPORTANT)
Create:

📄 src/pages/Home.jsx

jsx
Copy code
import { verses } from "../data/verses";
import { encouragements } from "../data/encouragements";
import VerseCard from "../components/VerseCard";

function getDailyItem(list) {
  const today = new Date().toDateString();
  let hash = 0;

  for (let i = 0; i < today.length; i++) {
    hash += today.charCodeAt(i);
  }

  return list[hash % list.length];
}

export default function Home() {
  const verse = getDailyItem(verses);
  const encouragement = getDailyItem(encouragements);

  return (
    <div style={styles.container}>
      <h1 style={styles.title}>Bénédictions Partagées</h1>
      <p style={styles.subtitle}>Today’s Word</p>

      <VerseCard verse={verse} />

      <div style={styles.encouragement}>
        <p>{encouragement}</p>
      </div>
    </div>
  );
}

const styles = {
  container: {
    padding: "20px",
    maxWidth: "500px",
    margin: "0 auto"
  },
  title: {
    textAlign: "center"
  },
  subtitle: {
    textAlign: "center",
    marginBottom: "20px"
  },
  encouragement: {
    background: "#eaf4ea",
    padding: "15px",
    borderRadius: "10px",
    textAlign: "center"
  }
};
